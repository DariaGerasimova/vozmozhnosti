  Функция ПолучитьКурсыВалют(ДатаКурса = Неопределено)

    Если ДатаКурса <> Неопределено Тогда
        СтрокаПараметров = "date_req=" + Формат(ДатаКурса, "ДФ=dd/MM/yyyy");

    Ответ = Клиент.Получить(Запрос);
    СтрокаXML = Ответ.ПолучитьТелоКакСтроку();
    ЧтениеXML = новый ЧтениеXML;
	ЧтениеXML.УстановитьСтроку(СтрокаXML);
	ДатаКурса = "";
    МассивКурсов = Новый Массив;
	пока ЧтениеXML.Прочитать() цикл
       Если ЧтениеXML.ТипУзла = ТипУзлаXML.НачалоЭлемента Тогда
                
                // Если нашли элемент Valute, начинаем сбор данных
                Если ЧтениеXML.Имя = "Valute" Тогда
                    // Сбрасываем данные для новой валюты
                    ТекущаяВалюта = "";
                    ТекущийКурс = "";
                    ТекущийНоминал = "";
                    ТекущееНаименование = "";
                КонецЕсли;
                
                // Читаем значения элементов внутри Valute
                Если ЧтениеXML.Имя = "CharCode" Тогда
                    ЧтениеXML.Прочитать();
                    Если ЧтениеXML.ТипУзла = ТипУзлаXML.Текст Тогда
                        ТекущаяВалюта = ЧтениеXML.Значение;
                    КонецЕсли;
                    
                ИначеЕсли ЧтениеXML.Имя = "Value" Тогда
                    ЧтениеXML.Прочитать();
                    Если ЧтениеXML.ТипУзла = ТипУзлаXML.Текст Тогда
                        ТекущийКурс = ЧтениеXML.Значение;
                    КонецЕсли;
                    
                ИначеЕсли ЧтениеXML.Имя = "Nominal" Тогда
                    ЧтениеXML.Прочитать();
                    Если ЧтениеXML.ТипУзла = ТипУзлаXML.Текст Тогда
                        ТекущийНоминал = ЧтениеXML.Значение;
                    КонецЕсли;
                    
                ИначеЕсли ЧтениеXML.Имя = "Name" Тогда
                    ЧтениеXML.Прочитать();
                    Если ЧтениеXML.ТипУзла = ТипУзлаXML.Текст Тогда
                        ТекущееНаименование = ЧтениеXML.Значение;
                    КонецЕсли;
                КонецЕсли;
                
            ИначеЕсли ЧтениеXML.ТипУзла = ТипУзлаXML.КонецЭлемента 
                И ЧтениеXML.Имя = "Valute" Тогда
                
                // Если закончился элемент Valute, добавляем данные в массив
                Если ТекущаяВалюта <> "" И ТекущийКурс <> "" Тогда
                    КурсВалюты = Новый Структура;
                    КурсВалюты.Вставить("Код", ТекущаяВалюта);          // Например: "USD", "EUR"
                    КурсВалюты.Вставить("Курс", ТекущийКурс);           // Например: "77,0540"
                    КурсВалюты.Вставить("Номинал", ТекущийНоминал);     // Например: "1", "100"
                    КурсВалюты.Вставить("Наименование", ТекущееНаименование); // Например: "Доллар США"
                    
                    МассивКурсов.Добавить(КурсВалюты);
				КонецЕсли;
			КонецЕсли;
			
		КонецЦикла;
		
		Возврат МассивКурсов;

Функция ИмпортЗаказовИзCSV(ПутьФайла) Экспорт
    
    ТекстФайла = новый ТекстовыйДокумент;
    ТекстФайла.Прочитать(ПутьФайла, КодировкаТекста.ANSI);
    
    Строки = СтрРазделить(ТекстФайла.ПолучитьТекст(), Символы.ПС, Ложь);
    
    // Если файл пустой
    Если Строки.Количество() = 0 Тогда
        Возврат новый Массив;
    КонецЕсли;
    
    Результат = новый Массив;
    
    // Определяем колонки по заголовку (первая строка)
    НазваниеЗаголовков = СтрРазделить(Строки[0], ";", Ложь);
    
    // Обрабатываем данные со второй строки
    Для Инд = 1 По Строки.Количество() - 1 Цикл
        Данные = СтрРазделить(Строки[Инд], ";", Ложь);
        
        Если Данные.Количество() > 0 Тогда
            Заказ = новый Структура;
            
            // Артикул (обычно первая колонка)
            Если НазваниеЗаголовков.Количество() > 0 Тогда
                ИмяКолонкиАртикул = НазваниеЗаголовков[0];
                Заказ.Вставить(ИмяКолонкиАртикул, Данные[0]);
            КонецЕсли;
            
            // Название (обычно вторая колонка)
            Если НазваниеЗаголовков.Количество() > 1 И Данные.Количество() > 1 Тогда
                ИмяКолонкиНазвание = НазваниеЗаголовков[1];
                Заказ.Вставить(ИмяКолонкиНазвание, Данные[1]);
            КонецЕсли;
            
            // Количество (обычно третья колонка)
            Если НазваниеЗаголовков.Количество() > 2 И Данные.Количество() > 2 Тогда
                ИмяКолонкиКоличество = НазваниеЗаголовков[2];
                Количество = 0;
                Попытка
                    Количество = Число(Данные[2]);
                Исключение
                    Количество = 0;
                КонецПопытки;
                Заказ.Вставить(ИмяКолонкиКоличество, Количество);
            КонецЕсли;
            
            Результат.Добавить(Заказ);
        КонецЕсли;
    КонецЦикла;
    
    Возврат Результат;
